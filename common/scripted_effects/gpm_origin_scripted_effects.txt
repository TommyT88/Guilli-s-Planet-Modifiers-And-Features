
# planet scope
gpm_run_origin_death_world_planet_gen = {
	optimize_memory
	
	## deposits ##	
	# clear all deposits so we can setup a harsh but curated start
	clear_deposits = yes
	
	# start with basic deposits to populate district options
	add_deposit = gpm_d_planetwide_jungles
	add_deposit = gpm_d_crystalized_hives
	add_deposit = gpm_d_carnivorous_jungle
	add_deposit = gpm_d_planetwide_forest
	add_deposit = gpm_d_super_volcano
	add_deposit = gpm_d_giant_skeleton
	add_deposit = d_tempestous_mountain
	add_deposit = d_ore_rich_caverns
	add_deposit = d_fungal_caves
	
	# deposits arachnophobia mode = off
	if = {
		limit = { has_global_flag = "gpm_spider_features_disabled" }
		add_deposit = gpm_d_massive_wildlife
		add_deposit = gpm_d_small_wildlife
		add_modifier = { modifier = "gpm_titanic_predators" }
	}
	
	# deposits arachnophobia mode = on
	if = {
		limit = { NOT = { has_global_flag = "gpm_spider_features_disabled" } }
		add_deposit = gpm_d_empty_nest
		add_deposit = gpm_d_arachnophobia
		add_modifier = { modifier = "gpm_arachnophobia" }
	}
	
	## planet modifiers ## 
	add_modifier = { modifier = "gpm_hostile_flora"  }
	add_modifier = { modifier = "gpm_carnivorous_flies" mult = 0.5 }
	
	# planet modifiers arachnophobia mode = off
	if = {
		limit = { has_global_flag = "gpm_spider_features_disabled" }
		add_modifier = { modifier = "gpm_titanic_predators" mult = 0.5 }
	}
	
	# planet modifiers arachnophobia mode = on
	if = {
		limit = { NOT = { has_global_flag = "gpm_spider_features_disabled" } }
		add_modifier = { modifier = "gpm_arachnophobia" mult = 0.5 }
	}
		
	# Add origin planet modifier
	add_modifier = { modifier = "gpm_death_world" }


	## blockers ##
	add_deposit = d_unsupervised_settlement
	add_deposit = d_ruined_building_blocker
	add_deposit = gpm_tb_swamp_resources
	add_deposit = gpm_tb_swamp_resources
	add_deposit = gpm_tb_lethal_overgrowth_resources
	add_deposit = gpm_tb_lethal_overgrowth_resources
	add_deposit = gpm_tb_lethal_overgrowth_resources
	add_deposit = d_dangerous_wildlife_blocker
	add_deposit = d_dangerous_wildlife_blocker
	add_deposit = gpm_tb_locust_plagues
	add_deposit = d_dense_jungle
	add_deposit = d_dense_jungle
	add_deposit = gpm_d_webbing
	add_deposit = gpm_d_webbing
	add_deposit = d_noxious_swamp
	add_deposit = d_venomous_insects
	add_deposit = d_rotten_soil
}
