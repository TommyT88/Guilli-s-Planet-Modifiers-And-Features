gpm_designate_wonder_world = {
	optimize_memory
	set_planet_flag = gpm_is_wondrous_world
	set_planet_flag = unique_world					# Real Space mod request
}

# desert
gpm_spawn_planet_wonders_desert = {
	optimize_memory
	random_list = {
		75 = {
			add_deposit = "gpm_d_gargantuan_sinkhole"
			add_modifier = { modifier = gpm_wonder_gargantuan_sinkhole }
			set_planet_flag = is_gpm_d_gargantuan_sinkhole_world
		}
		25 = {
			modifier = { factor = 0 
				AND = {
					has_global_flag = gpm_d_titanic_cave_wondrous_world_generated
					has_global_flag = gpm_d_abandoned_stargate_wondrous_world_generated
					has_global_flag = gpm_d_bioluminescent_forest_wondrous_world_generated
					has_global_flag = gpm_d_land_reefs_wondrous_world_generated
					has_global_flag = gpm_d_Collapsed_habitat_wondrous_world_generated
					has_global_flag = gpm_d_ruined_arcology_wondrous_world_generated
				}
			}
			gpm_spawn_planet_wonders_any = yes
		}
	}
}
