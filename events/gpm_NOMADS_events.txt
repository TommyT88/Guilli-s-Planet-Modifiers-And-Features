# Ark ship enters orbit of a HABITABLE world with regular planet modifiers
# After planet is surveyed
# While surveying: add effect in survey events

# FROM = planet
# This / controller = fleet
fleet_event = {
	id = gpm_NOMADS.1
	location = FROM
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		FROM = {
			is_scope_type = planet
			gpm_is_planet_class_habitable = yes
			gpm_has_precursor_modifiers = no
			gpm_has_NOMADS_surveyable_planet_modifiers = yes
			# Must be surveyed
			is_surveyed = {
				who = root.OWNER
				status = yes
			}
			# NOT = { has_modifier = gpm_ark_ship_visited }
			NOT = { has_planet_flag = gpm_nomad_visited@root.owner }
		}
		controller = {
			is_country_type = default
			is_nomadic = yes
		}
		is_arkship_fleet = yes
	}
	immediate = {
		# log =  "gpm_NOMADS.1 | root = [root.GetName] -- from = [from.GetName] -- PREV = [prev.GetName] -- this = [this.GetName]"
		begin_event_chain = {
			event_chain = "gpm_nomad_visit_chain"
			target = root.owner
		}
		enable_special_project = {
			name = "GPM_VISIT_WORLD_ARKSHIP"
			location = FROM
			owner = root.owner
		}
		# <TODO> add message at top telling you to visit!
	}
}
